cmake_minimum_required(VERSION 3.20)
project(zhitan_engine VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
include(CTest)
add_library(zhitan_direct src/direct_preview.cpp)
target_include_directories(zhitan_direct PUBLIC include)
if(MSVC)
  target_compile_options(zhitan_direct PRIVATE /W4)
else()
  target_compile_options(zhitan_direct PRIVATE -Wall -Wextra -Werror)
endif()
add_executable(direct_preview_test tests/direct_preview_test.cpp)
target_link_libraries(direct_preview_test PRIVATE zhitan_direct)
add_test(NAME direct_preview COMMAND direct_preview_test)
include(FetchContent)
set(JSON_BuildTests OFF CACHE BOOL "" FORCE)
FetchContent_Declare(nlohmann_json
  GIT_REPOSITORY https://github.com/nlohmann/json.git
  GIT_TAG 55f93686c01528224f448c19128836e7df245f72
)
FetchContent_MakeAvailable(nlohmann_json)
find_package(Threads REQUIRED)
add_executable(zhitan_worker src/main.cpp)
target_link_libraries(zhitan_worker PRIVATE zhitan_direct nlohmann_json::nlohmann_json Threads::Threads)
find_package(Python3 COMPONENTS Interpreter)
if(Python3_FOUND)
  add_test(NAME worker_protocol COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/worker_protocol_test.py $<TARGET_FILE:zhitan_worker>)
endif()

add_executable(job_state_test tests/job_state_test.cpp)
target_include_directories(job_state_test PRIVATE include)
target_link_libraries(job_state_test PRIVATE Threads::Threads)
add_test(NAME job_state COMMAND job_state_test)
set_tests_properties(worker_protocol PROPERTIES TIMEOUT 30)
